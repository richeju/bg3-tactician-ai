INITSECTION
KBSECTION
// Turning the ring on outside combat makes the actor's snapshot available immediately.
// In combat, BTAI_AdaptiveCombat scans when control is applied, before choosing tactics.
IF
StatusApplied((CHARACTER)_Actor,"BTAI_ENABLED",_,_)
AND
IsInCombat(_Actor,0)
THEN
PROC_BTAI_ScanCapabilities(_Actor);

IF
StatusRemoved((CHARACTER)_Actor,"BTAI_ENABLED",_,_)
THEN
PROC_BTAI_ClearCapabilities(_Actor);

IF
StatusRemoved((CHARACTER)_Actor,"BTAI_CONTROLLED",_,_)
AND
HasActiveStatus(_Actor,"BTAI_ENABLED",0)
THEN
PROC_BTAI_ClearCapabilities(_Actor);

// A delegated actor can carry both statuses through a save/load without an apply event.
IF
LevelGameplayStarted(_,_)
AND
DB_PartyMembers((CHARACTER)_Actor)
AND
HasActiveStatus(_Actor,"BTAI_ENABLED",1)
THEN
PROC_BTAI_ScanCapabilities(_Actor);

IF
LevelGameplayStarted(_,_)
AND
DB_BTAI_Scanned((CHARACTER)_Actor)
AND
HasActiveStatus(_Actor,"BTAI_ENABLED",0)
THEN
PROC_BTAI_ClearCapabilities(_Actor);

IF
Equipped(_,(CHARACTER)_Actor)
THEN
PROC_BTAI_RefreshEnabledCapabilities(_Actor);

IF
Unequipped(_,(CHARACTER)_Actor)
THEN
PROC_BTAI_RefreshEnabledCapabilities(_Actor);

IF
LeveledUp((CHARACTER)_Actor)
THEN
PROC_BTAI_RefreshEnabledCapabilities(_Actor);

IF
RespecCompleted((CHARACTER)_Actor)
THEN
PROC_BTAI_RefreshEnabledCapabilities(_Actor);

IF
LearnedSpell((CHARACTER)_Actor,_)
THEN
PROC_BTAI_RefreshEnabledCapabilities(_Actor);

PROC
PROC_BTAI_RefreshEnabledCapabilities((CHARACTER)_Actor)
AND
HasActiveStatus(_Actor,"BTAI_ENABLED",1)
THEN
PROC_BTAI_ScanCapabilities(_Actor);

PROC
PROC_BTAI_ScanCapabilities((CHARACTER)_Actor)
THEN
PROC_BTAI_ClearCapabilities(_Actor);
PROC_BTAI_CollectSpellRows(_Actor);
PROC_BTAI_ScanPassives(_Actor);
PROC_BTAI_ReadWeaponScores(_Actor);
DB_BTAI_Scanned(_Actor);

PROC
PROC_BTAI_CollectSpellRows((CHARACTER)_Actor)
AND
DB_BTAI_Catalog((STRING)_Spell,(STRING)_Category)
AND
HasSpell(_Actor,_Spell,1)
THEN
DB_BTAI_OwnedCapability(_Actor,_Spell,_Category);
PROC_BTAI_CheckPreparation(_Actor,_Spell,_Category);

// A class or equipment action can be owned without using the preparation mechanic.
// Keeping ownership separately also preserves the identity of known, unprepared spells.
PROC
PROC_BTAI_CheckPreparation((CHARACTER)_Actor,(STRING)_Spell,(STRING)_Category)
AND
IsSpellActive(_Actor,_Spell,1)
THEN
DB_BTAI_PreparedCapability(_Actor,_Spell,_Category);

PROC
PROC_BTAI_ClearCapabilities((CHARACTER)_Actor)
THEN
PROC_BTAI_ClearSpellRows(_Actor);
PROC_BTAI_ClearPassives(_Actor);
PROC_BTAI_ClearWeaponScores(_Actor);
NOT DB_BTAI_Scanned(_Actor);

PROC
PROC_BTAI_ClearSpellRows((CHARACTER)_Actor)
AND
DB_BTAI_OwnedCapability(_Actor,_Spell,_Category)
THEN
NOT DB_BTAI_OwnedCapability(_Actor,_Spell,_Category);

PROC
PROC_BTAI_ClearSpellRows((CHARACTER)_Actor)
AND
DB_BTAI_PreparedCapability(_Actor,_Spell,_Category)
THEN
NOT DB_BTAI_PreparedCapability(_Actor,_Spell,_Category);

EXITSECTION
ENDEXITSECTION
ParentTargetEdge "Start"
