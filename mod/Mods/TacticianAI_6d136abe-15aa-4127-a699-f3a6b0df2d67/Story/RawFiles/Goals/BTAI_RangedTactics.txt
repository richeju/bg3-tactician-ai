// Compare native scores when both main-hand slots contain a weapon.
QRY
QRY_BTAI_RangedBeatsMelee((CHARACTER)_Actor,(REAL)_RangedScore)
AND
GetEquippedItem(_Actor,"Melee Main Weapon",_Melee)
AND
IsWeapon(_Melee,1)
AND
GetWeaponScoreForCharacter(_Melee,_Actor,_MeleeScore)
AND
_MeleeScore > 0.0
AND
RealProduct(_MeleeScore,1.10,_Threshold)
AND
_RangedScore > _Threshold
THEN
DB_NOOP(1);

// An empty slot gives no weapon score. Keep trained unarmed fighters on their normal AI.
QRY
QRY_BTAI_RangedBeatsMelee((CHARACTER)_Actor,(REAL)_RangedScore)
AND
_RangedScore > 0.0
AND
NOT GetEquippedItem(_Actor,"Melee Main Weapon",_)
AND
HasPassive(_Actor,"MartialArts_UnarmedDamage",0)
AND
HasPassive(_Actor,"TavernBrawler",0)
THEN
DB_NOOP(1);

QRY
QRY_BTAI_PrefersRanged((CHARACTER)_Actor)
AND
GetEquippedItem(_Actor,"Ranged Main Weapon",_Ranged)
AND
IsRangedWeapon(_Ranged,0,1)
AND
IsProficientWith(_Actor,_Ranged,1)
AND
GetWeaponScoreForCharacter(_Ranged,_Actor,_RangedScore)
AND
QRY_BTAI_RangedBeatsMelee(_Actor,_RangedScore)
THEN
DB_NOOP(1);

// The shared delegation lifecycle calls this only for actors under its control.
PROC
PROC_BTAI_RefreshTactics((CHARACTER)_Actor)
AND
HasActiveStatus(_Actor,"BTAI_RANGED_TACTICS",0)
AND
QRY_BTAI_PrefersRanged(_Actor)
THEN
ApplyStatus(_Actor,"BTAI_RANGED_TACTICS",-1.0,1);

PROC
PROC_BTAI_RefreshTactics((CHARACTER)_Actor)
AND
HasActiveStatus(_Actor,"BTAI_RANGED_TACTICS",1)
AND
NOT QRY_BTAI_PrefersRanged(_Actor)
THEN
RemoveStatus(_Actor,"BTAI_RANGED_TACTICS");
